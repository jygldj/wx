改进四：修改 jiegu.html — 构建完整数据传给 AI
4.1 在 jieGuaBtn 点击事件中，构建 guaInfo 对象
在 jiegu.html 中找到 jieGuaBtn 的点击事件，在构建 guaInfo 的位置，替换为以下代码：

javascript
// 从 localStorage 读取用户信息
let userInfo = { name: '匿名', gender: '男', birth: '未知', question: '未记录' };
try {
    const raw = localStorage.getItem('userInfo');
    if (raw) userInfo = JSON.parse(raw);
} catch(e) {}

// 从卦象数据中获取起卦时间
let qiGuaTime = guaData?.time || new Date().toLocaleString();

// 构建详细的卦象数据（供 AI 直接使用）
const guaInfo = {
    // 基础信息
    benGua: guaData.benGuaName,
    benPalace: guaData.benPalace,
    bianGua: guaData.bianGuaName,
    bianPalace: guaData.bianPalace,
    dongYao: guaData.dongYao || [],
    yaoResults: guaData.yaoResults || [],
    dongStatus: guaData.dongStatus || [],
    
    // ⭐ 用户信息
    userName: userInfo.name,
    userGender: userInfo.gender,
    userBirth: userInfo.birth,
    userQuestion: userInfo.question,
    qiGuaTime: qiGuaTime,
    
    // ⭐ 世应位置（直接从排盘结果读取）
    shiYao: guaData.shiYao ? `第${guaData.shiYao}爻` + (guaData.benYao[guaData.shiYao - 1] ? 
        ` ${guaData.benYao[guaData.shiYao - 1].地支}${guaData.benYao[guaData.shiYao - 1].六亲}` : '') : '未知',
    yingYao: guaData.yingYao ? `第${guaData.yingYao}爻` + (guaData.benYao[guaData.yingYao - 1] ? 
        ` ${guaData.benYao[guaData.yingYao - 1].地支}${guaData.benYao[guaData.yingYao - 1].六亲}` : '') : '未知',
    
    // ⭐ 详细六爻数据（含动变详情）
    yaoDetail: guaData.benYao.map((yao, idx) => {
        const isDong = guaData.dongStatus && guaData.dongStatus[idx];
        const result = {
            dizhi: yao.地支,
            liuqin: yao.六亲,
            isDong: isDong || false
        };
        if (isDong && guaData.bianYao && guaData.bianYao[idx]) {
            result.bianDizhi = guaData.bianYao[idx].地支;
            result.bianLiuqin = guaData.bianYao[idx].六亲;
        }
        return result;
    }),
    
    // 用户问题
    question: question,
    
    // 时间信息
    timeInfo: timeInfo
};
